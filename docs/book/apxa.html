전략 --- 화면을 덮는 삼각형 하나

지금까지 우리는 물체를 이루는 삼각형 수백, 수천 개를 정점 버퍼에 담아 래스터화했다. 레이 트레이싱은 발상을 뒤집는다. 장면을 이루는 삼각형을 그리는 대신, 화면 전체를 덮는 삼각형 딱 하나를 그린다. 그러면 프래그먼트 셰이더가 화면의 모든 픽셀에서 한 번씩 실행되는데, 바로 그 픽셀 하나하나에서 카메라로부터 광선을 쏘아 장면과의 충돌을 계산한다. 물체는 정점이 아니라 셰이더 안의 수식(구의 방정식, 평면의 방정식 등)으로 존재한다.

화면을 덮는 데는 사각형(삼각형 2개) 대신 커다란 삼각형 하나를 쓰는 관용적 기법을 이용한다. 정점을 \((-1,-1),(3,-1),(-1,3)\)에 두면 이 삼각형은 클립 공간 \([-1,1]^2\)를 완전히 덮는다. 뼈대는 다음과 같다.

const canvas = document.getElementById('glCanvas');
const gl = canvas.getContext('webgl2', { antialias: true });

// 정점 셰이더: 화면을 덮는 삼각형을 그대로 통과시킨다
const vs = `#version 300 es
layout(location=0) in vec2 p;
void main(){ gl_Position = vec4(p, 0.0, 1.0); }`;

// 화면을 덮는 큰 삼각형 하나 (-1,-1) (3,-1) (-1,3)
const buf = gl.createBuffer();
gl.bindBuffer(gl.ARRAY_BUFFER, buf);
gl.bufferData(gl.ARRAY_BUFFER,
  new Float32Array([-1,-1, 3,-1, -1,3]), gl.STATIC_DRAW);
gl.enableVertexAttribArray(0);
gl.vertexAttribPointer(0, 2, gl.FLOAT, false, 0, 0);

// 프래그먼트 셰이더가 픽셀마다 광선을 쏜다 (아래 절에서 작성)
gl.drawArrays(gl.TRIANGLES, 0, 3);

이제 모든 일은 프래그먼트 셰이더 안에서 벌어진다.

카메라와 광선 생성

각 픽셀에서 가장 먼저 할 일은 그 픽셀을 통과하는 광선(광선의 시작점 ro와 방향 rd)을 만드는 것이다. gl_FragCoord로 픽셀 좌표를 얻어 \([-1,1]\) 범위의 화면 좌표 q로 바꾸고, 카메라 위치 eye에서 바라보는 지점 ta를 향하는 좌표계(오른쪽 rgt, 위 up, 앞 f)를 세운다. 표지는 세로로 긴 판형이므로, 가로 화각은 유지하고 세로만 늘리도록 종횡비를 처리한다.

vec2 q = (gl_FragCoord.xy / uRes) * 2.0 - 1.0;
float asp = uRes.x / uRes.y;
if (asp >= 1.0) q.x *= asp;   // 가로가 넓으면 가로 화각 확장
else            q.y /= asp;   // 세로 표지: 세로 화각 확장

vec3 eye = vec3(3.1, 1.85, 4.9);      // 카메라 위치
vec3 ta  = vec3(-0.05, 0.80, -0.10);  // 바라보는 지점
vec3 f   = normalize(ta - eye);
vec3 rgt = normalize(cross(f, vec3(0,1,0)));
vec3 up  = cross(rgt, f);
vec3 rd  = normalize(q.x*rgt + q.y*up + 2.0*f);  // 픽셀 광선 방향
vec3 ro  = eye;                                   // 광선 시작점

계수 2.0은 초점 거리에 해당한다. 값이 클수록 화각이 좁아져 망원처럼, 작을수록 광각처럼 보인다.

장면의 표현 --- 배열로 담은 물체들

장면의 모든 물체는 병렬 배열 몇 개로 표현한다. K는 모양(0 구, 1 정육면체, 2 원기둥, 3 팔면체, 4 다이아몬드), MT는 재질(0 유리, 1 메탈, 2 확산, 3 텍스처), C는 중심, S는 크기, T는 색(유리·메탈의 색조, 확산의 알베도)이다. 물체를 추가하거나 색을 바꾸려면 이 배열만 손대면 된다.

const int N = 20;                 // 물체 개수
int K[20]; int MT[20]; vec3 C[20]; vec3 S[20]; vec3 T[20];

void setup(){
  // 투명(유리) 물체 — K: 모양, MT=0: 유리
  K[0]=0; MT[0]=0; C[0]=vec3(-1.45,0.95,0.20); S[0]=vec3(0.95); T[0]=vec3(1.0,0.04,0.16); // 루비 구
  K[2]=4; MT[2]=0; C[2]=vec3( 2.00,0.55,-0.50);S[2]=vec3(0.72); T[2]=vec3(0.16,0.46,0.95);// 다이아몬드
  // 불투명 메탈 공 — MT=1
  K[6]=0; MT[6]=1; C[6]=vec3(-3.00,0.36,2.00); S[6]=vec3(0.36); T[6]=vec3(1.0,0.08,0.10); // 빨강 메탈
  // 텍스처 지구본 — MT=3
  K[12]=0; MT[12]=3; C[12]=vec3(-1.50,0.60,2.15); S[12]=vec3(0.60); T[12]=vec3(0.2,0.4,0.6);
  // 배경의 불투명 구·정육면체 — MT=1(메탈) 또는 2(확산)
  K[13]=1; MT[13]=2; C[13]=vec3(-1.20,0.60,-3.00); S[13]=vec3(0.60); T[13]=vec3(0.12,0.30,0.85);
  // …나머지 물체 생략 (같은 방식)

  // 광원 세 개: 위치 LP, 색·세기 LC
  LP[0]=vec3(5.0,7.5,4.0);  LC[0]=vec3(1.0,0.96,0.88)*1.15;
  LP[1]=vec3(-5.0,5.5,1.5); LC[1]=vec3(0.6,0.78,1.0)*0.9;
  LP[2]=vec3(1.0,4.5,-5.0); LC[2]=vec3(1.0,0.68,0.44)*0.85;
}

광선-물체 교차

광선 \(\mathbf{o}+t\,\mathbf{d}\)가 물체와 만나는 가장 가까운 거리 \(t\)와 그 지점의 법선을 구하는 것이 레이 트레이싱의 핵심 계산이다. 구는 2차 방정식으로 정확히 풀린다.

// 광선과 구의 교차: 가장 가까운 t, 표면 법선 n을 돌려준다
float iSphere(vec3 ro, vec3 rd, vec3 c, float r, out vec3 n){
  vec3 oc = ro - c;
  float b = dot(oc, rd);
  float k = dot(oc, oc) - r*r;
  float h = b*b - k;                 // 판별식
  if (h < 0.0) return -1.0;          // 만나지 않음
  h = sqrt(h);
  float t = -b - h;                  // 가까운 근
  if (t < 1e-3) t = -b + h;          // 안에서 출발했으면 먼 근
  if (t < 1e-3) return -1.0;
  n = normalize((ro + rd*t) - c);
  return t;
}

표지의 주인공인 크라운컷 다이아몬드는 평면(반평면) 여러 개의 교집합으로 정의되는 볼록 다면체다. 위쪽의 평평한 테이블 면, 비스듬한 크라운 패싯 8개, 아래로 뾰족하게 모이는 파빌리온 패싯 8개를 각각 반평면 \(\mathbf{n}\!\cdot\!\mathbf{p}\le d\)로 두고, 광선을 이 반평면들로 차례로 잘라(슬랩 클리핑) 들어오고 나가는 구간을 구한다.

// 크라운컷 다이아몬드: 테이블 + 크라운/파빌리온 패싯의 볼록 다면체
float iGem(vec3 ro, vec3 rd, vec3 c, float s, out vec3 n){
  vec3 o = ro - c;
  float tN = -1e9, tF = 1e9;           // 들어오는 구간 [tN, tF]
  vec3 nEnter = vec3(0,1,0), nExit = vec3(0,1,0);

  // 반평면 하나로 광선 구간을 자르는 매크로 같은 처리 (테이블·크라운·파빌리온에 반복)
  //  fn: 면 법선, d: 면까지 거리 → 영역은 dot(fn, p) <= d
  //  den>0 이면 나가는 면(tF 갱신), den<0 이면 들어오는 면(tN·법선 갱신)
  for (int i = 0; i < 8; i++){
    float a = 6.2831853 * float(i) / 8.0;
    vec3 fn = normalize(vec3(cos(a), 0.85, sin(a)));   // 크라운 패싯(위-바깥)
    float d = dot(fn, vec3(s*cos(a), 0.0, s*sin(a)));
    float den = dot(fn, rd), num = d - dot(fn, o);
    if (abs(den) < 1e-7){ if (num < 0.0) return -1.0; }
    else { float t = num/den;
      if (den < 0.0){ if (t > tN){ tN = t; nEnter = fn; } }
      else          { if (t < tF){ tF = t; nExit  = fn; } } }
    // 파빌리온 패싯(아래-바깥, 22.5° 회전)도 같은 방식으로 잘라낸다 …
  }
  if (tN > tF || tF < 1e-3) return -1.0;
  if (tN > 1e-3){ n = nEnter; return tN; }  // 바깥에서 진입
  n = nExit; return tF;                      // 안에서 탈출
}

정육면체(iBox)는 슬랩 3쌍, 원기둥·팔면체도 같은 요령의 해석적 교차로 구현한다. scene() 함수는 바닥 평면과 모든 물체에 대해 교차를 시도하고 가장 가까운 것을 고른다.

색을 결정하는 바운스 루프

광선이 무엇과 만났는지 알았으면, 그 지점의 색을 정한다. 유리와 금속은 빛을 다시 튕기거나 통과시키므로, 광선을 여러 번 바운스시키며 색을 누적한다. thr(throughput)은 지금까지 광선이 잃지 않고 나른 빛의 비율이다.

vec3 col = vec3(0.0), thr = vec3(1.0);
for (int bounce = 0; bounce < 7; bounce++){
  float t; vec3 n; int oi;
  int which = scene(ro, rd, t, n, oi);
  if (which == -1){ col += thr * sky(rd);            break; } // 하늘
  if (which ==  0){ col += thr * shadeFloor(ro+rd*t, rd); break; } // 바닥

  vec3 p = ro + rd*t; vec3 v = -rd;
  if (MT[oi] == 3){ col += thr*shadeDiffuse(p,n,v,texGlobe(n)); break; } // 지구본
  if (MT[oi] == 2){ col += thr*shadeDiffuse(p,n,v,T[oi]);       break; } // 확산
  if (MT[oi] == 1){                                     // 불투명 메탈: 반사만
    vec3 nf = dot(rd,n)<0.0 ? n : -n;
    float ci = max(dot(v,nf), 0.0);
    vec3 F = T[oi] + (1.0 - T[oi]) * pow(1.0-ci, 5.0); // 색조 프레넬
    thr *= F;  rd = reflect(rd, nf);  ro = p + rd*1e-3;
    continue;
  }
  // 유리(유전체): 반사 + 굴절
  bool entering = dot(rd,n) < 0.0;  vec3 nf = entering ? n : -n;
  float ior = (K[oi]==4) ? 2.42 : 1.5;                 // 다이아몬드는 높은 굴절률
  float ci = max(dot(v,nf), 0.0);
  float fres = 0.04 + 0.96*pow(1.0-ci, 5.0);           // 프레넬 반사율
  col += thr * fres * /* 반사 방향의 환경색 */ ;
  vec3 rt = refract(rd, nf, entering ? 1.0/ior : ior); // 굴절 방향
  thr *= (1.0-fres) * mix(vec3(1.0), T[oi], entering ? 0.30 : 0.0); // 색조 흡수
  rd = rt;  ro = p + rd*1e-3;
}
col *= 0.66;   // 노출

프레넬Fresnel 항 fres는 스치는 각도에서 반사가 강해지는 현상을 근사한다. 금속은 굴절 없이 반사만 하되 반사색을 금속색 T로 물들이고, 유리는 반사와 굴절로 빛을 나눈다. 다이아몬드는 굴절률을 2.42로 높여 내부 반사가 화려하게 일어난다.

다이아몬드의 색분산(무지개 파이어)은 패싯의 법선과 시선으로 색상(hue)을 뽑아 코사인 팔레트로 무지개를 만들고, 프레넬 엣지로 가중해 얹는다. 값싼 연산 한 줄로 보석의 반짝임을 흉내 낸다.

if (K[oi] == 4){   // 다이아몬드 색분산 — 패싯별 무지개 파이어
  float edge = 0.22 + 0.78*pow(1.0-ci, 2.0);
  float hue  = fract(dot(nf, vec3(1.7,0.9,1.3))*0.5 + dot(nf,v)*0.6);
  vec3  fire = 0.5 + 0.5*cos(6.2831853*(hue + vec3(0.0, 0.33, 0.67)));
  col += thr * (1.0-fres) * edge * fire * 0.95;
}

배경 --- 하늘, 구름, 바닥

광선이 아무 물체도 맞히지 못하면 하늘이다. 시선의 높이에 따라 짙은 남색에서 지평선의 옅은 파랑으로 섞고, fbm 잡음을 하늘 평면에 투영해 구름을 얹는다.

vec3 sky(vec3 rd){
  float h = clamp(rd.y*0.5 + 0.5, 0.0, 1.0);
  vec3 c = mix(vec3(0.09,0.17,0.36), vec3(0.002,0.02,0.12), pow(h, 0.62));
  if (rd.y > 0.015){                       // 지평선 위에만 구름
    vec2 cp = rd.xz / rd.y * 1.15;          // 하늘 평면에 투영
    float f = fbm(cp*0.52 + vec2(3.0,1.5));
    float cov = smoothstep(0.46, 0.86, f);
    vec3 cloud = mix(vec3(0.20,0.25,0.38), vec3(0.48,0.52,0.62),
                     smoothstep(0.4, 0.95, f));
    c = mix(c, cloud, cov * smoothstep(0.02,0.28,rd.y) * 0.85);
  }
  return c;
}

바닥은 무한 체커 평면이다. 멀어질수록 칸이 촘촘해져 생기는 계단 잡음을 막기 위해, 화소가 덮는 폭 fwidth로 체커를 해석적으로 안티에일리어싱한다.

vec3 checkerCol(vec2 p){
  vec2 w = fwidth(p) + 1e-3;             // 이 화소가 덮는 바닥 폭
  vec2 i = 2.0*(abs(fract((p-0.5*w)*0.5)-0.5)
              - abs(fract((p+0.5*w)*0.5)-0.5)) / w;
  float c = 0.5 - 0.5*i.x*i.y;           // 경계에서 부드럽게 평균
  return mix(vec3(0.04,0.06,0.14), vec3(0.55,0.58,0.66), c);
}

그림자와 caustic

바닥의 각 점에서 광원으로 그림자 광선을 쏘아 물체가 가리면 어둡게 한다(shadowT). 유리는 완전히 막지 않고 빛을 조금 통과시키므로 그림자를 옅게, 불투명 물체는 짙게 감쇠한다. 여기에 굴절 caustic을 더한다. 광원에서 물체 중심을 지나 바닥에 맺히는 지점을 초점으로 삼아, 물체 색을 띤 밝은 링과 핫스팟을 부드러운 그림자 안쪽에 얹어, 유리가 빛을 모아 만든 밝은 무늬를 흉내 낸다.

화면 밖에서 완성하기 --- FBO 블룸 후처리

여기까지가 장면을 한 번 그리는 과정이다. 마지막으로 11장에서 배운 프레임버퍼 오브젝트framebuffer object(FBO)를 써서 블룸(bloom)을 입힌다. 블룸은 밝은 부분의 빛이 번져 나오는 효과로, 보석의 하이라이트와 색분산, caustic을 은은하게 발광시켜 장면을 극적으로 만든다. 구조는 화면이 아니라 텍스처에 그린 뒤 그 텍스처를 다시 입력으로 쓰는 2패스(여기서는 4패스) 포스트 프로세싱이다.

  1. 장면 패스: 레이 트레이서를 화면이 아니라 부동소수점 텍스처(RGBA16F)에 그린다. 밝기가 1을 넘는 하이라이트를 손실 없이 보존하기 위해서다.
  2. 밝은 부분 추출: 그 텍스처에서 임계값보다 밝은 화소만 골라 반해상도 텍스처에 담는다.
  3. 가우시안 블러: 추출한 밝은 부분을 가로·세로로 분리해 흐린다. 두 개의 FBO를 번갈아 쓰는 핑퐁으로 여러 번 반복해 넓게 번지게 한다.
  4. 합성: 원본 장면에 흐린 밝은 부분을 더하고, 비네트·감마·대비·채도를 적용해 화면에 출력한다.

먼저 후처리 셰이더 세 편이다. 밝은 부분 추출과 분리형 블러, 그리고 합성이다.

// (2) 밝은 부분만 추출
uniform sampler2D uTex; uniform float uThreshold;
void main(){
  vec3 c = texture(uTex, uv).rgb;
  float l = max(max(c.r, c.g), c.b);
  o = vec4(c * smoothstep(uThreshold, uThreshold+0.6, l), 1.0);
}
// (3) 분리형 가우시안 블러 (uDir로 가로/세로 선택)
uniform vec2 uTexel; uniform vec2 uDir;
void main(){
  vec2 d = uDir * uTexel * 1.4;
  vec3 s = texture(uTex, uv).rgb * 0.2270;
  s += (texture(uTex,uv+d).rgb    + texture(uTex,uv-d).rgb)    * 0.1946;
  s += (texture(uTex,uv+2.0*d).rgb+ texture(uTex,uv-2.0*d).rgb)* 0.1216;
  s += (texture(uTex,uv+3.0*d).rgb+ texture(uTex,uv-3.0*d).rgb)* 0.0541;
  s += (texture(uTex,uv+4.0*d).rgb+ texture(uTex,uv-4.0*d).rgb)* 0.0162;
  o = vec4(s, 1.0);
}
// (4) 합성 + 그레이딩
uniform sampler2D uScene, uBloom; uniform float uBloomInt;
void main(){
  vec3 c = texture(uScene,uv).rgb + texture(uBloom,uv).rgb * uBloomInt;
  float dc = length(uv-0.5);  c *= clamp(1.0 - 1.6*dc*dc, 0.16, 1.0); // 비네트
  c = pow(clamp(c,0.0,1.0), vec3(0.4545));                            // 감마
  c = clamp((c-0.5)*1.20 + 0.5, 0.0, 1.0);                            // 대비
  float lum = dot(c, vec3(0.299,0.587,0.114));
  o = vec4(clamp(mix(vec3(lum), c, 1.22), 0.0, 1.0), 1.0);            // 채도
}

JavaScript 쪽에서는 렌더 타깃 텍스처와 FBO를 만들고, 네 패스를 순서대로 그린다. 색 버퍼로 부동소수점 텍스처를 쓰기 위해 EXT_color_buffer_float 확장을 켠다.

gl.getExtension('EXT_color_buffer_float');   // 부동소수점 렌더 타깃 허용

// 렌더 타깃 텍스처(RGBA16F)와 FBO를 만드는 헬퍼
function mkTex(w, h){
  const t = gl.createTexture();
  gl.bindTexture(gl.TEXTURE_2D, t);
  gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA16F, w, h, 0,
                gl.RGBA, gl.HALF_FLOAT, null);   // 빈 부동소수점 텍스처
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
  return t;
}
function mkFBO(t){
  const f = gl.createFramebuffer();
  gl.bindFramebuffer(gl.FRAMEBUFFER, f);
  gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT0,
                          gl.TEXTURE_2D, t, 0);   // 색 버퍼 = 이 텍스처
  return f;
}

직접 해보기

전체 코드는 하나의 HTML 파일이므로, 브라우저에서 열어 곧바로 값을 바꿔 볼 수 있다.

  • setup()의 배열에 물체를 추가하거나 T[] 색을 바꿔 자신만의 장면을 꾸며 보자.
  • 합성 패스의 uThreshold(블룸 임계값)와 uBloomInt(세기), col *= 0.66(노출)을 조절해 분위기를 바꿔 보자.
  • 블러 반복 횟수를 늘리면 더 넓게 번지고, uTexel 계수를 키우면 더 성기게 번진다.
  • 다이아몬드 색분산의 fire 강도(* 0.95)를 올려 더 화려한 보석을 만들어 보자.

한 장의 삼각형에서 시작해, 광선과 재질, 절차적 배경, 그리고 화면 밖 렌더링까지 — 이 책이 지나온 길이 이 한 장의 표지에 고스란히 담겨 있다.

   float cov=smoothstep(0.46,0.86,f);
   float fade=smoothstep(0.02,0.28,rd.y)*(1.0-0.30*h);
   vec3 cloud=mix(vec3(0.20,0.25,0.38),vec3(0.48,0.52,0.62),smoothstep(0.4,0.95,f));
   c=mix(c,cloud,clamp(cov*fade*0.85,0.0,1.0));
 }
 for(int i=0;i<NL;i++) c+=LC[i]*pow(max(dot(rd,normalize(LP[i])),0.0),160.0)*0.7;
 return c;}
vec3 texGlobe(vec3 nrm){
 float u=atan(nrm.z,nrm.x); float vv=asin(clamp(nrm.y,-1.0,1.0));
 vec2 uv=vec2(u,vv);
 float f=fbm(uv*vec2(1.5,3.0)+vec2(2.0,0.0));
 float land=smoothstep(0.53,0.585,f);
 vec3 ocean=mix(vec3(0.03,0.20,0.52),vec3(0.06,0.34,0.62),smoothstep(0.3,0.53,f));
 vec3 green=mix(vec3(0.13,0.42,0.16),vec3(0.48,0.40,0.20),smoothstep(0.60,0.85,f));
 vec3 col=mix(ocean,green,land);
 float ice=smoothstep(1.15,1.42,abs(vv)); col=mix(col,vec3(0.94,0.96,0.98),ice);
 return col;}
vec3 checkerCol(vec2 p){
 vec2 w=fwidth(p)+1e-3;
 vec2 i=2.0*(abs(fract((p-0.5*w)*0.5)-0.5)-abs(fract((p+0.5*w)*0.5)-0.5))/w;
 float c=0.5-0.5*i.x*i.y;
 return mix(vec3(0.04,0.06,0.14), vec3(0.55,0.58,0.66), c);
}
float shadowT(vec3 ro,vec3 rd,float maxt){ float s=1.0;
 for(int i=0;i<N;i++){ vec3 nn; float ti=iObj(i,ro,rd,nn); if(ti>0.0&&ti<maxt){ s*=(MT[i]==0)?0.48:0.12; } } return s;}
vec3 highlights(vec3 p,vec3 nf,vec3 v,float shin,float k){
 vec3 sp=vec3(0.0);
 for(int i=0;i<NL;i++){ vec3 L=normalize(LP[i]-p); vec3 hh=normalize(L+v);
   sp += LC[i]*pow(max(dot(nf,hh),0.0),shin); }
 return sp*k;
}
vec3 shadeDiffuse(vec3 p,vec3 nf,vec3 v,vec3 alb){
 vec3 lit=vec3(0.10)+sky(nf)*0.10;
 for(int i=0;i<NL;i++){ vec3 L=normalize(LP[i]-p); float df=max(dot(nf,L),0.0);
   float sh=(i==0)?shadowT(p+nf*1e-3,L,length(LP[i]-p)):1.0; lit+=LC[i]*df*sh; }
 return alb*lit + highlights(p,nf,v,45.0,0.28);
}
vec3 shadeFloor(vec3 p,vec3 rd){ vec3 n=vec3(0,1,0); vec3 base=checkerCol(p.xz); vec3 v=-rd;
 vec3 lit=vec3(0.0);
 for(int i=0;i<NL;i++){ vec3 L=normalize(LP[i]-p); float df=max(dot(n,L),0.0);
   float sh=(i==0)?shadowT(p+n*1e-3,L,length(LP[i]-p)):1.0; lit+=LC[i]*df*sh; }
 vec3 caus=vec3(0.0);
 for(int i=0;i<N;i++){
   if(MT[i]!=0) continue;
   vec3 dir=C[i]-LP[0];
   float s=LP[0].y/(LP[0].y-C[i].y);
   vec2 sc=(LP[0]+s*dir).xz;
   float R=objRad(i)*1.25;
   float r=length(p.xz-sc);
   float ring=exp(-pow((r-R*0.50)/(R*0.15),2.0));
   float core=exp(-pow(r/(R*0.32),2.0));
   caus += (T[i]*0.7+vec3(0.3)) * (ring*1.30+core*0.75);
 }
 vec3 col=base*(0.06+0.52*lit) + caus*(0.40+0.60*base)*1.5 + highlights(p,n,v,60.0,0.22);
 return mix(col,sky(reflect(rd,n)),0.05);}
void main(){
 setup();
 vec2 q=(gl_FragCoord.xy/uRes)*2.0-1.0;
 float asp=uRes.x/uRes.y;
 if(asp>=1.0) q.x*=asp; else q.y/=asp;
 vec3 eye=vec3(3.1,1.85,4.9), ta=vec3(-0.05,0.80,-0.10);
 vec3 f=normalize(ta-eye), rgt=normalize(cross(f,vec3(0,1,0))), up=cross(rgt,f);
 vec3 rd=normalize(q.x*rgt+q.y*up+2.0*f), ro=eye;
 vec3 col=vec3(0.0), thr=vec3(1.0);
 for(int bounce=0;bounce<7;bounce++){
   float t; vec3 n; int oi; int which=scene(ro,rd,t,n,oi);
   if(which==-1){ col+=thr*sky(rd); break; }
   if(which==0){ col+=thr*shadeFloor(ro+rd*t,rd); break; }
   vec3 p=ro+rd*t; vec3 v=-rd;
   if(MT[oi]==3){
     vec3 nf=dot(rd,n)<0.0?n:-n;
     col += thr*shadeDiffuse(p,nf,v,texGlobe(nf));
     break;
   }
   if(MT[oi]==2){
     vec3 nf=dot(rd,n)<0.0?n:-n;
     col += thr*shadeDiffuse(p,nf,v,T[oi]);
     break;
   }
   if(MT[oi]==1){
     vec3 nf=dot(rd,n)<0.0?n:-n;
     col += thr*highlights(p,nf,v,140.0,0.85);
     float ci=max(dot(v,nf),0.0);
     vec3 F=T[oi]+(1.0-T[oi])*pow(1.0-ci,5.0);
     col += thr*T[oi]*0.05;
     thr *= F;
     rd=reflect(rd,nf); ro=p+rd*1e-3;
     if(max(thr.r,max(thr.g,thr.b))<0.02) break;
     continue;
   }
   bool entering=dot(rd,n)<0.0; vec3 nf=entering?n:-n;
   float ior=(K[oi]==4)?2.42:IOR;
   float ci=max(dot(v,nf),0.0); float fres=0.04+0.96*pow(1.0-ci,5.0);
   col+=thr*highlights(p,nf,v,220.0,0.9);
   if(K[oi]==4){
     float edge=0.22+0.78*pow(1.0-ci,2.0);
     float hue=fract(dot(nf,vec3(1.7,0.9,1.3))*0.5 + dot(nf,v)*0.6 + dot(p,vec3(2.1,1.4,1.7))*0.12);
     vec3 fire=0.5+0.5*cos(6.2831853*(hue+vec3(0.0,0.33,0.67)));
     col += thr*(1.0-fres)*edge*fire*0.95;
   }
   vec3 rr=reflect(rd,nf); float rt; vec3 rn; int roi; int rw=scene(p+nf*1e-3,rr,rt,rn,roi);
   vec3 refl = rw==-1?sky(rr):(rw==0?shadeFloor(p+rr*rt,rr):(MT[roi]==0?T[roi]*0.8+0.2:T[roi]*0.72+0.12));
   col+=thr*fres*refl;
   float eta=entering?1.0/ior:ior;
   vec3 rt2=refract(rd,nf,eta);
   if(dot(rt2,rt2)<1e-5){ rd=reflect(rd,nf); ro=p+rd*1e-3; thr*=0.94; continue; }
   thr*=(1.0-fres)*mix(vec3(1.0),T[oi],entering?(K[oi]==4?0.55:0.30):0.0);
   rd=rt2; ro=p+rd*1e-3;
   if(max(thr.r,max(thr.g,thr.b))<0.02) break;
 }
 col*=0.66;                                  // 노출
 outColor=vec4(max(col,0.0),1.0);            // 선형 HDR → 후처리(블룸/톤맵)로 전달
}`;
//============================ 2) 밝은 부분 추출 (bright pass) ============================
const fsBright=`#version 300 es
precision highp float;
out vec4 o; uniform sampler2D uTex; uniform vec2 uRes; uniform float uThreshold;
void main(){
 vec2 uv=gl_FragCoord.xy/uRes;
 vec3 c=texture(uTex,uv).rgb;
 float l=max(max(c.r,c.g),c.b);
 float t=smoothstep(uThreshold,uThreshold+0.6,l);
 o=vec4(c*t,1.0);
}`;
//============================ 3) 분리형 가우시안 블러 ============================
const fsBlur=`#version 300 es
precision highp float;
out vec4 o; uniform sampler2D uTex; uniform vec2 uRes; uniform vec2 uTexel; uniform vec2 uDir;
void main(){
 vec2 uv=gl_FragCoord.xy/uRes;
 vec2 d=uDir*uTexel*1.4;
 float w0=0.227027,w1=0.1945946,w2=0.1216216,w3=0.054054,w4=0.016216;
 vec3 s=texture(uTex,uv).rgb*w0;
 s+=(texture(uTex,uv+d*1.0).rgb+texture(uTex,uv-d*1.0).rgb)*w1;
 s+=(texture(uTex,uv+d*2.0).rgb+texture(uTex,uv-d*2.0).rgb)*w2;
 s+=(texture(uTex,uv+d*3.0).rgb+texture(uTex,uv-d*3.0).rgb)*w3;
 s+=(texture(uTex,uv+d*4.0).rgb+texture(uTex,uv-d*4.0).rgb)*w4;
 o=vec4(s,1.0);
}`;
//============================ 4) 합성 + 톤맵 + 그레이딩 ============================
const fsComp=`#version 300 es
precision highp float;
out vec4 o; uniform sampler2D uScene; uniform sampler2D uBloom; uniform vec2 uRes; uniform float uBloomInt;
void main(){
 vec2 uv=gl_FragCoord.xy/uRes;
 vec3 scn=texture(uScene,uv).rgb;
 vec3 blm=texture(uBloom,uv).rgb;
 vec3 c=scn+blm*uBloomInt;                    // 블룸 합성 (어두운 장면은 그대로, 밝은 곳만 발광)
 float dc=length(uv-0.5); c*=clamp(1.0-1.6*dc*dc,0.16,1.0);  // 비네트
 c=pow(clamp(c,0.0,1.0),vec3(0.4545));         // 감마
 c=clamp((c-0.5)*1.20+0.5,0.0,1.0);            // 대비
 float lum=dot(c,vec3(0.299,0.587,0.114));
 c=clamp(mix(vec3(lum),c,1.22),0.0,1.0);       // 채도
 o=vec4(c,1.0);
}`;

function sh(t,s){const o=gl.createShader(t);gl.shaderSource(o,s);gl.compileShader(o);
 if(!gl.getShaderParameter(o,gl.COMPILE_STATUS)){document.title='ERR';console.log(gl.getShaderInfoLog(o));}return o;}
function prog(v,f){const p=gl.createProgram();gl.attachShader(p,sh(gl.VERTEX_SHADER,v));gl.attachShader(p,sh(gl.FRAGMENT_SHADER,f));gl.linkProgram(p);
 if(!gl.getProgramParameter(p,gl.LINK_STATUS)){document.title='LINKERR';console.log(gl.getProgramInfoLog(p));}return p;}
const pScene=prog(vs,fsScene), pBright=prog(vs,fsBright), pBlur=prog(vs,fsBlur), pComp=prog(vs,fsComp);
const buf=gl.createBuffer();gl.bindBuffer(gl.ARRAY_BUFFER,buf);
gl.bufferData(gl.ARRAY_BUFFER,new Float32Array([-1,-1,3,-1,-1,3]),gl.STATIC_DRAW);
gl.enableVertexAttribArray(0);gl.vertexAttribPointer(0,2,gl.FLOAT,false,0,0);
function mkTex(w,h){const t=gl.createTexture();gl.bindTexture(gl.TEXTURE_2D,t);
 gl.texImage2D(gl.TEXTURE_2D,0,gl.RGBA16F,w,h,0,gl.RGBA,gl.HALF_FLOAT,null);
 gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_MIN_FILTER,gl.LINEAR);
 gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_MAG_FILTER,gl.LINEAR);
 gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_WRAP_S,gl.CLAMP_TO_EDGE);
 gl.texParameteri(gl.TEXTURE_2D,gl.TEXTURE_WRAP_T,gl.CLAMP_TO_EDGE);return t;}
function mkFBO(t){const f=gl.createFramebuffer();gl.bindFramebuffer(gl.FRAMEBUFFER,f);
 gl.framebufferTexture2D(gl.FRAMEBUFFER,gl.COLOR_ATTACHMENT0,gl.TEXTURE_2D,t,0);return f;}
const W=canvas.width,H=canvas.height, hw=W>>1, hh=H>>1;
const sceneT=mkTex(W,H), sceneF=mkFBO(sceneT);
const bA=mkTex(hw,hh), bAF=mkFBO(bA);
const bB=mkTex(hw,hh), bBF=mkFBO(bB);
function U(p,n){return gl.getUniformLocation(p,n);}
function draw(){gl.drawArrays(gl.TRIANGLES,0,3);}
// ---- 패스 1: 장면 → sceneT (선형 HDR) ----
gl.bindFramebuffer(gl.FRAMEBUFFER,sceneF); gl.viewport(0,0,W,H);
gl.useProgram(pScene); gl.uniform2f(U(pScene,'uRes'),W,H); draw();

gl.useProgram(pBlur); gl.uniform2f(U(pBlur,'uTexel'),1.0/hw,1.0/hh); gl.uniform2f(U(pBlur,'uRes'),hw,hh);
gl.uniform1i(U(pBlur,'uTex'),0);
for(let i=0;i<3;i++){
 // 수평: bA → bB
 gl.bindFramebuffer(gl.FRAMEBUFFER,bBF); gl.viewport(0,0,hw,hh);
 gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D,bA); gl.uniform2f(U(pBlur,'uDir'),1.0,0.0); draw();
 // 수직: bB → bA
 gl.bindFramebuffer(gl.FRAMEBUFFER,bAF); gl.viewport(0,0,hw,hh);
 gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D,bB); gl.uniform2f(U(pBlur,'uDir'),0.0,1.0); draw();
}

gl.bindFramebuffer(gl.FRAMEBUFFER,null); gl.viewport(0,0,W,H);
