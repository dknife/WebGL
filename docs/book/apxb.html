APPENDIX B

WebGL 2.0 고급 기능

이 장을 마치면

이 부록에서 다루는 것

  • 유니폼 버퍼 오브젝트(UBO)로 유니폼을 묶어 여러 프로그램이 공유하기
  • 트랜스폼 피드백으로 정점 셰이더 출력을 버퍼로 되돌려 GPU 파티클 만들기
  • 다중 렌더 타깃(MRT)으로 한 번에 여러 텍스처에 출력하기(지연 셰이딩의 토대)

이 책의 본문은 WebGL 2.0의 기능 중 입문에 꼭 필요한 것들을 썼다. 여기서는 본문에서 노트로만 언급한 세 가지 저수준 기능을 참조용으로 정리한다. 셋 다 "GPU에게 데이터를 주고받는 방식"을 넓혀, 더 크고 효율적인 렌더링을 가능하게 한다.

유니폼 버퍼 오브젝트(UBO)

지금까지 유니폼은 uniform1f·uniformMatrix4fv로 하나씩 넘겼다. 뷰·투영 행렬처럼 여러 프로그램이 똑같이 쓰는 유니폼을 프로그램마다 다시 넘기는 것은 낭비다. 유니폼 버퍼 오브젝트uniform buffer object(UBO)는 유니폼들을 하나의 버퍼(블록)에 담아 두고, 여러 프로그램이 그 블록을 공유하게 한다. 카메라가 움직이면 UBO 하나만 갱신하면 모든 셰이더가 새 행렬을 본다.

// 셰이더: 유니폼 블록으로 선언
layout(std140) uniform Camera {
  mat4 uView;
  mat4 uProj;
};
// JS: 버퍼를 바인딩 포인트 0에 묶고, 프로그램의 블록을 그 포인트에 연결
const ubo = gl.createBuffer();
gl.bindBuffer(gl.UNIFORM_BUFFER, ubo);
gl.bufferData(gl.UNIFORM_BUFFER, 128, gl.DYNAMIC_DRAW);   // mat4 두 개 = 128바이트
gl.bindBufferBase(gl.UNIFORM_BUFFER, 0, ubo);
const idx = gl.getUniformBlockIndex(program, 'Camera');
gl.uniformBlockBinding(program, idx, 0);
// 카메라가 바뀌면 이 버퍼만 갱신하면 모든 프로그램이 공유
gl.bufferSubData(gl.UNIFORM_BUFFER, 0, viewMatrix);
gl.bufferSubData(gl.UNIFORM_BUFFER, 64, projMatrix);

std140 레이아웃은 블록 안 변수들의 메모리 정렬 규칙이다. vec3이 16바이트 경계에 맞춰지는 등 규칙이 있어, JS 쪽에서 오프셋을 맞출 때 주의해야 한다.

트랜스폼 피드백

정점 셰이더의 출력은 늘 래스터화로 흘러갔다. 트랜스폼 피드백transform feedback은 그 출력을 화면이 아니라 버퍼로 되돌려 받는다. 정점 셰이더를 "버퍼 \(\to\) 계산 \(\to\) 버퍼"의 병렬 계산기로 쓰는 셈이다. 10.6절의 파티클을 CPU가 아니라 GPU에서 갱신하는 것이 대표적 용도다.

// 출력으로 되돌릴 varying을 지정하고 링크
gl.transformFeedbackVaryings(program, ['vNewPos', 'vNewVel'], gl.SEPARATE_ATTRIBS);
gl.linkProgram(program);
// 갱신 패스: 래스터화는 끄고, 결과 버퍼로 캡처
gl.enable(gl.RASTERIZER_DISCARD);
gl.bindBufferBase(gl.TRANSFORM_FEEDBACK_BUFFER, 0, outPosBuf);
gl.beginTransformFeedback(gl.POINTS);
gl.drawArrays(gl.POINTS, 0, N);       // 정점 셰이더가 새 위치를 계산해 버퍼에 씀
gl.endTransformFeedback();
gl.disable(gl.RASTERIZER_DISCARD);

보통 입력 버퍼와 출력 버퍼를 매 프레임 맞바꾸는 핑퐁으로, 수십만 파티클을 CPU 왕복 없이 GPU 안에서 갱신한다.

다중 렌더 타깃(MRT)

프래그먼트 셰이더가 색 하나만 출력하던 제약을 푸는 것이 다중 렌더 타깃multiple render targets(MRT)이다. 하나의 FBO에 색 텍스처 여러 장을 붙이고, 셰이더에서 layout(location=N) out으로 각각에 동시에 쓴다.

layout(location=0) out vec4 gPosition;   // 위치
layout(location=1) out vec4 gNormal;     // 법선
layout(location=2) out vec4 gAlbedo;     // 재질색
// FBO에 텍스처 셋을 각각 다른 attachment에 붙이고
gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT0, gl.TEXTURE_2D, texPos, 0);
gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT1, gl.TEXTURE_2D, texNrm, 0);
gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT2, gl.TEXTURE_2D, texAlb, 0);
gl.drawBuffers([gl.COLOR_ATTACHMENT0, gl.COLOR_ATTACHMENT1, gl.COLOR_ATTACHMENT2]);

이렇게 화면의 위치·법선·재질을 미리 여러 텍스처(G-buffer)에 그려 두고, 조명을 그다음에 화면 공간에서 한 번에 계산하는 것이 지연 셰이딩deferred shading이다(11.6절). 광원이 수백 개인 장면에서, 광원마다 전체 장면을 다시 그리는 대신 화면 픽셀당 한 번씩만 조명을 계산할 수 있다.

요점

세 기능의 공통점은 "GPU와 데이터를 주고받는 통로를 넓힌다"는 것이다 — UBO는 유니폼을 묶어 공유하고, 트랜스폼 피드백은 정점 출력을 버퍼로 되받고, MRT는 프래그먼트 출력을 여러 텍스처로 내보낸다. 이 통로들이 대규모 파티클·지연 셰이딩·GPU 시뮬레이션의 토대가 되며, WebGPU(13.3절)는 이 방향을 더 밀고 나간 것이다.

전체 코드 — 트랜스폼 피드백 GPU 파티클

이 부록에서 다룬 세 기능 가운데 트랜스폼 피드백을 처음부터 끝까지 돌려 보는 완결 예제다. 파티클 3000개의 위치 — 속도 — 수명을 모두 GPU의 정점 셰이더에서 갱신해 버퍼로 되돌리고(핑퐁), 그 결과를 점 스프라이트로 그린다. CPU는 매 프레임 uDt 하나만 넘길 뿐이다. 웹 에디션에서 [실행]으로 분수가 솟는 것을 확인해 보자.

// 트랜스폼 피드백 GPU 파티클 — 갱신을 전부 GPU에서(정점 셰이더 출력을 버퍼로 되돌림)
const gl = glCanvas.getContext('webgl2', { antialias: true });
const N = 3000;

// --- 갱신 프로그램: (pos, vel, life) → (newPos, newVel, newLife) ---
const updVS = `#version 300 es
layout(location=0) in vec2 aPos;
layout(location=1) in vec2 aVel;
layout(location=2) in float aLife;
uniform float uDt; uniform float uSeed;
out vec2 vPos; out vec2 vVel; out float vLife;
float hash(float n){ return fract(sin(n)*43758.5453); }
void main(){
  float life = aLife - uDt;
  vec2 pos = aPos, vel = aVel;
  if(life <= 0.0){                                  // 재생성 (바닥 분출)
    float s = float(gl_VertexID) + uSeed;
    float a = hash(s)*6.2831;
    float sp = 0.4 + hash(s+1.3)*0.5;
    pos = vec2(0.0, -0.9);
    vel = vec2(cos(a)*0.25, 0.9 + hash(s+2.7)*0.6);
    life = 0.7 + hash(s+3.1)*0.8;
  } else {
    vel.y -= 1.3*uDt;                               // 중력
    pos += vel*uDt;
  }
  vPos = pos; vVel = vel; vLife = life;
}`;
const updFS = `#version 300 es
precision highp float; void main(){}`;   // 래스터화 안 함

// --- 렌더 프로그램: 점 스프라이트 + 가산 블렌딩 ---
const renVS = `#version 300 es
layout(location=0) in vec2 aPos;
layout(location=2) in float aLife;
out float vLife;
void main(){ vLife=aLife; gl_Position=vec4(aPos,0.0,1.0); gl_PointSize=4.0+aLife*14.0; }`;
const renFS = `#version 300 es
precision highp float; in float vLife; out vec4 o;
void main(){ vec2 d=gl_PointCoord-0.5; float r=1.0-smoothstep(0.0,0.5,length(d));
  vec3 c=mix(vec3(1.0,0.3,0.05), vec3(0.5,0.85,1.0), vLife); o=vec4(c*r, r); }`;

function sh(t,s){ const o=gl.createShader(t); gl.shaderSource(o,s); gl.compileShader(o);
  if(!gl.getShaderParameter(o,gl.COMPILE_STATUS)) console.log(gl.getShaderInfoLog(o)); return o; }
function prog(vs,fs,tfVaryings){
  const p=gl.createProgram();
  gl.attachShader(p, sh(gl.VERTEX_SHADER, vs));
  gl.attachShader(p, sh(gl.FRAGMENT_SHADER, fs));
  if(tfVaryings) gl.transformFeedbackVaryings(p, tfVaryings, gl.INTERLEAVED_ATTRIBS);
  gl.linkProgram(p);
  if(!gl.getProgramParameter(p,gl.LINK_STATUS)) console.log(gl.getProgramInfoLog(p));
  return p;
}
const updP = prog(updVS, updFS, ['vPos','vVel','vLife']);
const renP = prog(renVS, renFS, null);

// 초기 파티클 데이터 (pos.xy, vel.xy, life) = 5 float/입자
const init = new Float32Array(N*5);
for(let i=0;i<N;i++){
  const a=Math.random()*6.28;
  init[i*5]=0; init[i*5+1]=-0.9;
  init[i*5+2]=Math.cos(a)*0.25; init[i*5+3]=0.9+Math.random()*0.6;
  init[i*5+4]=Math.random()*1.2;                    // 흩어진 초기 수명
}
// 핑퐁 버퍼 두 벌
function makeBuf(data){ const b=gl.createBuffer(); gl.bindBuffer(gl.ARRAY_BUFFER,b);
  gl.bufferData(gl.ARRAY_BUFFER, data, gl.DYNAMIC_COPY); return b; }
let bufA = makeBuf(init), bufB = makeBuf(new Float32Array(N*5));
const STRIDE = 5*4;
function bindAttribs(buf){
  gl.bindBuffer(gl.ARRAY_BUFFER, buf);
  gl.enableVertexAttribArray(0); gl.vertexAttribPointer(0,2,gl.FLOAT,false,STRIDE,0);   // pos
  gl.enableVertexAttribArray(1); gl.vertexAttribPointer(1,2,gl.FLOAT,false,STRIDE,8);   // vel
  gl.enableVertexAttribArray(2); gl.vertexAttribPointer(2,1,gl.FLOAT,false,STRIDE,16);  // life
}
const tf = gl.createTransformFeedback();

let last=0;
function frame(t){
  const dt = last ? Math.min((t-last)/1000, 0.05) : 0.016; last=t;
  // 1) 갱신 패스: bufA를 읽어 bufB에 새 상태를 쓴다 (래스터화 끔)
  gl.useProgram(updP);
  gl.uniform1f(gl.getUniformLocation(updP,'uDt'), dt);
  gl.uniform1f(gl.getUniformLocation(updP,'uSeed'), (t%1000)*0.7);
  bindAttribs(bufA);
  gl.bindTransformFeedback(gl.TRANSFORM_FEEDBACK, tf);
  gl.bindBufferBase(gl.TRANSFORM_FEEDBACK_BUFFER, 0, bufB);
  gl.enable(gl.RASTERIZER_DISCARD);
  gl.beginTransformFeedback(gl.POINTS);
  gl.drawArrays(gl.POINTS, 0, N);
  gl.endTransformFeedback();
  gl.disable(gl.RASTERIZER_DISCARD);
  gl.bindBufferBase(gl.TRANSFORM_FEEDBACK_BUFFER, 0, null);
  gl.bindTransformFeedback(gl.TRANSFORM_FEEDBACK, null);
  // 2) 렌더 패스: 방금 쓴 bufB를 그린다
  gl.useProgram(renP);
  bindAttribs(bufB);
  gl.enable(gl.BLEND); gl.blendFunc(gl.SRC_ALPHA, gl.ONE);
  gl.clearColor(0.02,0.02,0.05,1); gl.clear(gl.COLOR_BUFFER_BIT);
  gl.drawArrays(gl.POINTS, 0, N);
  // 3) 버퍼 교체 (핑퐁)
  const tmp=bufA; bufA=bufB; bufB=tmp;
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
